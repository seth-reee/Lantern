cmake_minimum_required(VERSION 3.24)
project(Lantern VERSION 0.1.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.8 REQUIRED COMPONENTS Quick QuickControls2 Network)
qt_policy(SET QTP0001 NEW)
qt_policy(SET QTP0004 NEW)
add_library(scanner src/scanner.cpp src/scanner.h)
target_include_directories(scanner PUBLIC src)
target_link_libraries(scanner PUBLIC Qt6::Network Qt6::Gui)
qt_add_executable(lantern src/main.cpp src/theme.h)
target_compile_definitions(lantern PRIVATE LANTERN_VERSION="${PROJECT_VERSION}")
qt_add_resources(lantern about_assets PREFIX "/" FILES Lantern.png)
qt_add_qml_module(lantern URI Lantern VERSION 1.0 QML_FILES qml/Main.qml)
target_link_libraries(lantern PRIVATE scanner Qt6::Quick Qt6::QuickControls2)
install(TARGETS lantern RUNTIME DESTINATION bin)
install(FILES packaging/lantern.desktop DESTINATION share/applications)
install(FILES Lantern.png DESTINATION share/pixmaps RENAME lantern.png)
install(FILES LICENSE DESTINATION share/licenses/lantern)
include(CTest)
if(BUILD_TESTING)
 find_package(Qt6 REQUIRED COMPONENTS Test)
 qt_add_executable(scanner_tests tests/scanner_test.cpp)
 target_link_libraries(scanner_tests PRIVATE scanner Qt6::Test)
 add_test(NAME scanner COMMAND scanner_tests)
 qt_add_executable(ui_tests tests/ui_test.cpp src/theme.h)
 qt_add_resources(ui_tests about_assets PREFIX "/" FILES Lantern.png)
 target_link_libraries(ui_tests PRIVATE scanner Qt6::Test Qt6::Quick Qt6::QuickControls2)
 target_compile_definitions(ui_tests PRIVATE LANTERN_QML="${CMAKE_CURRENT_SOURCE_DIR}/qml/Main.qml")
 add_test(NAME ui_results COMMAND ui_tests)
 set_tests_properties(ui_results PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QPA_PLATFORMTHEME=basic;QT_QUICK_BACKEND=software")
 add_test(NAME ui_smoke COMMAND lantern --smoke-test)
 set_tests_properties(ui_smoke PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QPA_PLATFORMTHEME=basic;QT_QUICK_BACKEND=software")
endif()
